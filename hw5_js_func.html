Напишите функцию, которая возвращает меньшее из двух чисел. Если числа равны — возвращает это значение. 

// Задание 1: Меньшее из двух чисел

    if (a <= b) {
        return a;
    } else {
        return b;
    }
}

Приветствие по возрасту

Создайте функцию, которая спрашивает возраст и выводит соответствующее приветствие. 

// Задание 4: Приветствие по возрасту
function greetByAge() {
    let age = parseInt(prompt("Сколько вам лет?"));

    if (age < 0) {
        alert("Вы ввели неправильное значение");
    } else if (age >= 0 && age <= 12) {
        alert("Привет, друг!");
    } else {
        alert("Добро пожаловать!");
    }
}

Проверка чисел и произведение

Напишите функцию, которая проверяет, являются ли оба параметра числами, и возвращает их произведение. 

// Задание 5: Проверка чисел и произведение
function multiplyNumbers(a, b) {
    if (isNaN(a) || isNaN(b)) {
        return "Одно или оба значения не являются числом";
    }
    return a * b;
}

Задание 6

Куб числа

Напишите функцию, которая запрашивает число, проверяет его и возвращает строку с кубом числа. 

// Задание 6: Куб числа
function cubeNumber() {
    let n = prompt("Введите число");
    if (isNaN(n) || n === "" || n === null) {
        return "Переданный параметр не является числом";
    }
    return n + " в кубе равняется " + (n ** 3);
}
